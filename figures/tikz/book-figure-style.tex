% Shared visual vocabulary for the book's original diagrams.
% All diagrams inherit these fonts, strokes and semantic styles.

\definecolor{BookGold}{HTML}{B17A00}
\definecolor{BookTeal}{HTML}{247C86}
\definecolor{BookInk}{HTML}{30363B}
\definecolor{BookLight}{HTML}{E9EFF5}

\providecommand{\BookTinyFont}{\scriptsize\rmfamily}

  book factor/.style={rectangle, draw=BookBlue, fill=BookBlue, minimum size=2.3mm, inner sep=0pt, outer sep=0pt},
  book measurement factor/.style={book factor, draw=BookGold, fill=BookGold},
  book odometry/.style={draw=BookBlue, book stroke},
  book measurement/.style={draw=BookGold, dashed, book stroke},
  book figure/.style={font=\BookFigureFont, color=BookBlue, text=BookInk,
    line width=0.8pt, >=Latex},
  book axis/.style={->, draw=BookInk, line width=0.6pt},
  book link/.style={->, draw=BookBlue, line width=0.8pt},
  book observation/.style={draw=BookGold, fill=BookGold!8},
  book state/.style={draw=BookBlue, fill=BookLight},
  book note/.style={font=\BookNoteFont, text=BookInk}
