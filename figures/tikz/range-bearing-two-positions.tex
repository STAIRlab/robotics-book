\begin{document}
\begin{tikzpicture}[book figure, scale=1.05,>=Latex]

\coordinate (L)  at (0,0);
\coordinate (X1) at (4,2);
\coordinate (X2) at (2.5,3);

% Radii in cm (robust)
\pgfmathsetmacro{\rOne}{veclen(4,2)}
\pgfmathsetmacro{\rTwo}{veclen(2.5,3)}

% Bearings
\pgfmathanglebetweenpoints{\pgfpointanchor{L}{center}}{\pgfpointanchor{X1}{center}}
\let\bearingOne\pgfmathresult
\pgfmathanglebetweenpoints{\pgfpointanchor{L}{center}}{\pgfpointanchor{X2}{center}}
\let\bearingTwo\pgfmathresult

% Axes
\draw[book axis] (-0.5,0) -- (5.5,0) node[right] {$x$};
\draw[book axis] (0,-0.5) -- (0,5.2) node[above] {$y$};

% Landmark + robots
\fill[BookGold] (L) circle (2pt) node[below left] {Landmark};
\fill[BookBlue] (X1) circle (2pt) node[right, xshift=3pt, yshift=-1pt] {$\mathbf{x}_1$};
\fill[BookTeal] (X2) circle (2pt) node[above, xshift=-1pt, yshift=3pt] {$\mathbf{x}_2$};

% Range circles
\draw[BookBlue,dashed] (L) circle (\rOne cm);
\draw[BookTeal,dash pattern=on 4pt off 2pt on 1pt off 2pt] (L) circle (\rTwo cm);

% Range lines
\draw[BookBlue,book curve] (L)--(X1);
\draw[BookTeal,book curve] (L)--(X2);

% Bearing rays
\draw[BookBlue,book curve,->] (L)--($(L)!1.25!(X1)$);
\draw[BookTeal,book curve,->] (L)--($(L)!1.25!(X2)$);

% Angle arcs
\draw[BookBlue] ($(L)+(1.2,0)$) arc (0:\bearingOne:1.2);
\draw[BookTeal] ($(L)+(1.7,0)$) arc (0:\bearingTwo:1.7);

\node at ($(L)+(0.9,0.22)$) {$\theta_1$};
\node at ($(L)+(1.1,0.9)$) {$\theta_2$};

\end{tikzpicture}
