\begin{tikzpicture}[book figure, scale=1.05,>=Latex]
    \coordinate (L) at (0,0);
    \coordinate (X) at (4,2);

    % Compute bearing angle (degrees) from L to X
    \pgfmathanglebetweenpoints{\pgfpointanchor{L}{center}}{\pgfpointanchor{X}{center}}
    \let\bearingAngle\pgfmathresult

    % Point along the bearing ray: extend segment L->X by 25%
    \coordinate (R) at ($(L)!1.25!(X)$);

    % Axes
    \draw[book axis] (-0.5,0) -- (6,0) node[right] {$x$};
    \draw[book axis] (0,-0.5) -- (0,3.5) node[above] {$y$};

    % Landmark and robot
    \fill[BookGold] (L) circle (2pt) node[below left] {Landmark $\ell=(\ell_x,\ell_y)$};
    \fill[BookBlue] (X) circle (2pt) node[right=8mm,text width=3.5cm,align=left,fill=white,inner sep=2pt] {Robot $\mathbf{x}=(x,y)=(4,2)$ [Example]};

    % Range line
    \draw[BookBlue,book curve] (L) -- (X);
    \node at (2.1,1.2) {$r$};

    % Bearing ray
    \draw[BookBlue,book curve,->] (L) -- (R);

    % Angle arc using computed end angle
    \draw[BookGold,book stroke] ($(L)+(0.9,0)$) arc[start angle=0,end angle=\bearingAngle,radius=0.9];
    \node at ($(L)+(2.6,0.3)$) {$\theta = \pgfmathprintnumber[precision=2]{\bearingAngle}^\circ$ [Example]};

    % Optional dashed circle (illustrative)
    \draw[BookTeal,dashed,book stroke] (L) circle (4.47);

    \node[text=BookTeal] at (-1.7,2.9) {\small Range locus: circle};
    \node[text=BookBlue] at (5.3,3.6) {\small Bearing locus: ray ($\theta$)};
  \end{tikzpicture}
